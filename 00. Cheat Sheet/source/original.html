10 + 3             # 13       10 - 3  → 7
10 * 3             # 30       10 / 4  → 2.5
10 // 3            # 3        10 % 3  → 1
2 ** 3             # 8

total = 0
total += 5         # add and save

04Strings & slicing

word = "foundation"
word[0]                # "f"
word[-1]               # "n"
word[:2]               # "fo"
word[-2:]              # "on"
word[2:5]              # "und"

05Clean & combine text

text = "  ANNA  "
text.strip()               # "ANNA"
text.strip().capitalize()  # "Anna"
text.lower()               # "  anna  "
text.upper()               # "  ANNA  "
"one fine day".split()     # list of words
"Java".replace("Java", "Python")
"@" in "a@b.com"           # True
"a@b.com".find("@")        # 1

words = ("one", "fine", "day")
" ".join(words)            # "one fine day"
"-".join(words)            # "one-fine-day"

Strings are immutable: methods return new strings. strip(chars) removes matching edge characters, not an exact word.

06Lists & tuples

numbers = [10, 20, 30]          # mutable list
numbers.append(40)              # add one item
numbers.insert(0, 5)            # insert at position
numbers.remove(20)              # remove by value
last = numbers.pop()            # remove + return last

alias = numbers                 # same list
copy = numbers.copy()           # new, shallow list

person = ("Anna", "Stockholm")  # tuple
name, city = person             # unpack
first, *middle, last = [1, 2, 3, 4]
# first=1, middle=[2, 3], last=4

Tuples cannot have items replaced. A shallow copy still shares nested objects.

07Dictionaries

laptop = {"name": "Acer", "price": 250}
laptop["price"]            # 250
laptop["price"] = 300      # update
laptop["colour"] = "grey"  # add
laptop.pop("colour")       # remove + return

laptop.get("ram")          # None if missing

laptop.keys()              # keys

08Sets & nested data

a = {"python", "git"}
b = {"git", "sql"}
a & b                    # shared: {"git"}
a | b                    # union: all skills
a - b                    # only in a: {"python"}
a.add("docker")
a.discard("missing")     # no error if absent
# a.remove("missing")    # KeyError if absent
set([1, 1, 2])           # {1, 2}

courses = [{"topics": ["HTML", "CSS"]}]
courses[0]["topics"][1]  # "CSS"

Sets keep unique values; do not rely on their display order. Chained indexing moves one level at a time.

09Conditions

score = 75
active = True

if active and score >= 70:
    print("PASS")
elif not active:
    print("Inactive")
else:
    print("FAIL")

# and: both       or: at least one
# not: reverse the truth value
# False, None, 0, "", [], {}, set() are falsy

Indentation defines the block. Only the first matching if/elif branch runs.

10Loops & totals

numbers = [4, 12, 7, 3, 9]
total = 0
for number in numbers:
    total += number
print(total)           # 35; after the loop

for n in range(1, 4):  # 1, 2, 3
    print(n)
# range(5): 0–4; range(10, 0, -1): 10–1

for row in range(5):
    for column in range(5):
        print("x", end=" ")
    print()            # end this row

The loop variable is ONE item; the collection contains ALL items. Start the total before the loop.

11Functions & return

def calculate_total(numbers):
    total = 0
    for number in numbers:
        total += number
    return total  # after the whole loop

result = calculate_total([4, 12, 7])
print(result)     # 23
# or: print(calculate_total([4, 12, 7]))

# calculate_total       → function itself
# calculate_total(...)  → call, get its result

return sends a VALUE back and exits the function. print displays a value. No return value → None.

12Defaults & scope

def greet(name, greeting="Hello"):
    return f"{greeting} {name}"

greet("Anna")                      # default
greet("Anna", "Hej")               # positional
greet(greeting="Hi", name="Anna")  # keywords

def outer():
    message = "outside"
    def inner():
        print(message)             # enclosing scope
    inner()

Parameters/local variables belong to the function call. Use descriptive names; avoid list, str, sum, max as variable names.

13*args: positional arguments

def add_all(*numbers):  # collect → tuple
    total = 0
    for number in numbers:
        total += number
    return total

add_all(10, 20, 30)     # separate arguments
values = [10, 20, 30]
add_all(*values)        # unpack list → arguments

# def function(*args): collects
# function(*values):   unpacks

For an average: total / len(numbers). Decide how to handle no arguments before dividing.

14**kwargs: keyword arguments

def create_user(username, **details):
    return {"username": username, **details}

create_user("mao", city="Stockholm")
# {"username": "mao", "city": "Stockholm"}

settings = {"city": "Stockholm"}
create_user("mao", **settings)  # unpack in call

# def function(**kwargs): collect → dictionary
# {"details": details}:   nested dictionary
# {**details}:            unpack pairs into dict

Normal parameters are separate from **kwargs. With only **kwargs, you can return kwargs directly.

15Filter & find

def keep_settings(**settings):
    kept = {}
    for key, value in settings.items():
        if value is not None:
            kept[key] = value
    return kept

def longest_word(*words):
    longest = ""
    for word in words:
        if len(word) > len(longest):
            longest = word
    return longest

Save the best value, not just its length. Return after checking all items. None, "None", and "" are different.

17Sorting & lambda

numbers = [3, 1, 2]
ordered = sorted(numbers)  # new list
numbers.sort()             # change list; returns None

players = [{"name": "Anna", "score": 85}]
ranked = sorted(
    players,
    key=lambda player: player["score"],
    reverse=True           # highest first
)

# lambda parameter: expression

key chooses the value used for comparison. The sorted result still contains whole player dictionaries.

18enumerate & zip

names = ["Sara", "Emma", "Anna"]
scores = [94, 88, 85]

for rank, name in enumerate(names, start=1):
    print(rank, name)

for name, score in zip(names, scores):
    print(name, score)

pairs = list(zip(names, scores))
# [("Sara", 94), ("Emma", 88), ("Anna", 85)]

enumerate → number + item. zip → corresponding items; stops at the shortest input. A zip iterator is consumed as you use it.

19Classes & __init__

class Book:
    def __init__(self, title, author, pages):
        self.title = title
        self.author = author
        self.pages = pages

book = Book("The Hobbit", "J.R.R. Tolkien", 310)
# Book       → class
# book       → one object (instance)
# book.title → an attribute on that object

self is the current object. self.title stores the supplied title on it. __init__ sets up the new object; it must not return a dictionary or another value.

20Read & change attributes

class Laptop:
    def __init__(self, brand, ram_gb, price):
        self.brand = brand
        self.ram_gb = ram_gb
        self.price = price

laptop1 = Laptop("HP", 16, 10000)
laptop2 = Laptop("Apple", 8, 30000)

print(laptop1.price)  # 10000
laptop1.price = 8000  # update this object
print(laptop1.price)  # 8000
print(laptop2.price)  # 30000

Use object.attribute for objects; dictionary["key"] for dictionaries. Choose the attribute you mean to change. Store prices and RAM as numbers when you need calculations.

21Lists of objects & printing

# Using the Book class from section 19:
book1 = Book("The Hobbit", "J.R.R. Tolkien", 310)
book2 = Book("1984", "George Orwell", 328)
books = [book1, book2]

for book in books:
    print(
        f"Title: {book.title}",
        f"Author: {book.author}",
        f"Pages: {book.pages}",
        sep="\n"
    )
    print()            # blank line between books

book1.pages = 320
print(books[0].pages)  # 320: same object

The list holds references to your objects, not copies. In the loop, book is one object at a time. sep separates print arguments; end follows the last argument. Source-code line breaks alone do not make output line breaks.

22Same values vs same object

class Horse:
    def __init__(self, color):
        self.color = color

horse1 = Horse("brown")
horse2 = Horse("brown")

print(horse1.color == horse2.color)  # True
print(horse1 is horse2)              # False

same_horse = horse1
print(same_horse is horse1)          # True
same_horse.color = "white"
print(horse1.color)                  # white

Each Horse(...) creates a separate object. is checks identity. Assigning an object to another variable gives it another name; it does not make a copy.

23Defaults & keyword arguments

class Clothes:
    def __init__(self, shirt, pants="blue"):
        self.shirt = shirt
        self.pants = pants

clothes1 = Clothes("green", "black")  # positional
clothes2 = Clothes("green")           # default pants
clothes3 = Clothes(
    pants="black", shirt="green"      # keyword arguments
)

print(clothes1.pants)                 # black
print(clothes2.pants)                 # blue

Defaults apply when an argument is omitted. Keyword arguments name the parameter, so these can be supplied in a different order. self is passed automatically; you do not supply it in Clothes(...).

24Shared collection of instances

class Laptop:
    all_laptops = []  # shared class attribute

    def __init__(self, brand, price):
        self.brand = brand
        self.price = price
        Laptop.all_laptops.append(self)

laptop1 = Laptop("HP", 10000)
laptop2 = Laptop("Apple", 30000)

for laptop in Laptop.all_laptops:
    print(laptop.brand, laptop.price)

Optional pattern we discussed: each object adds itself to a shared list. This list keeps the objects until you remove them. A separate laptops = [...] list is often simpler. For a separate list per object, create self.items = [] inside __init__.

25Instance methods & self

class Book:
    def __init__(self, title, pages):
        self.title = title
        self.pages = pages

    def is_long(self):  # aligned with __init__
        return self.pages > 300

book = Book("The Hobbit", 310)
print(book.pages)       # attribute → 310
print(book.is_long())   # call method → True

check = book.is_long    # method, not its result
print(check())          # call it later → True

A method is a function defined in a class. Calling book.is_long() supplies book as self automatically. self.pages reads that book’s page count. Both def lines belong inside the class; neither method belongs inside the other.

26Return a comparison

# Two ways to write the same method:
class Book:
    def __init__(self, pages):
        self.pages = pages

    def is_long(self):
        if self.pages > 300:
            return True
        else:
            return False

    def is_long_short(self):
        return self.pages > 300

print(Book(300).is_long())        # False
print(Book(301).is_long_short())  # True

The comparison already produces a boolean. Returning it directly is equivalent to the full if/else. A simplification suggestion from the editor does not mean the longer version is wrong. Without a return on the false path, that path would return None.

28Mutable defaults: shared list

class BadTeam:
    def __init__(self, name, members=[]):
        self.name = name
        self.members = members

    def add_member(self, name):
        self.members.append(name)

red = BadTeam("Red")
blue = BadTeam("Blue")
red.add_member("Anna")
print(blue.members)                 # ["Anna"]
print(red.members is blue.members)  # True

A default argument is evaluated once when the function is defined, not once per call. Both teams reuse the same default list. append changes it in place. The usual fix is a None default and creating a fresh list inside __init__ when no list was supplied.

29Inheritance & super()

class Animal:
    def __init__(self, name):
        self.name = name

    def make_sound(self) -> str:
        return "some animal sound"

class Dog(Animal):
    def __init__(self, name, breed):
        super().__init__(name)
        self.breed = breed

dog = Dog("Rex", "terrier")
print(dog.name)          # Rex
print(dog.make_sound())  # some animal sound

Dog is a subclass of Animal: a dog is an animal. super() reuses the next implementation in the method resolution order; here that is Animal. If Dog defines no __init__, it inherits the base initializer. Writing a new __init__ does not automatically call the base one.

30Override & extend a method

class Employee:
    def get_information(self) -> str:
        return "Employee information"

class Developer(Employee):
    def get_information(self) -> str:
        base = super().get_information()
        return base + " - Role: Developer"

employee = Employee()
developer = Developer()
print(employee.get_information())
print(developer.get_information())

Defining the same method name in a subclass overrides the inherited version. super() lets the override reuse it. -> str declares the intended return type; it does not convert the result. Calling the method on an Employee still uses Employee’s implementation.

31Change state & raise an error

class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner
        self.balance = balance

    def deposit(self, amount):
        self.balance += amount

    def withdraw(self, amount):
        if amount < 0 or amount > self.balance:
            raise ValueError("Invalid withdrawal")
        self.balance -= amount

account = BankAccount("Anna", 100)
account.deposit(50)
account.withdraw(20)
print(account.balance)  # 130

Check before changing state. -= subtracts and saves; = -amount replaces the balance with a negative value. raise signals failure; return ValueError would only return the exception class. These methods change the object and return None. This small example validates withdrawals only.

32Polymorphism & duck typing

class Printer:
    def display_status(self):
        return "Printer ready"

class Screen:
    def display_status(self):
        return "Screen on"

things = [Printer(), Screen()]
for thing in things:
    print(thing.display_status())

One call, different implementations: polymorphism. Duck typing means this code relies on the supported operation, not a shared base class. Python tries the call; an object without the method would raise AttributeError. No custom __init__ is needed here.

33Inherited setup, overridden method

class Document:
    def __init__(self, title):
        self.title = title

    def describe(self) -> str:
        return f"Document: {self.title}"

class PDFDocument(Document):
    def describe(self) -> str:
        return f"PDF: {self.title}"

docs = [Document("Notes"), PDFDocument("Guide")]
for doc in docs:
    print(doc.describe())

PDFDocument inherits the title setup because it defines no __init__. It overrides describe instead. self.title reads the stored title; self.title = ... would change it. Returning a string describes the object without changing its state.

34pass, return & optional __init__

class User:
    pass

class AdminUser(User):
    pass

admin = AdminUser()  # an instance

def example():
    pass             # do nothing; continue
    return "Hello"   # exit; send a value back

print(example())     # Hello

pass fills an otherwise empty block. return exits a function; a bare return gives None. __init__ supplies custom setup and is not required in every class. An empty subclass inherits its initializer; an empty override would skip the inherited setup unless it calls super().

35isinstance: type relationships

class Animal:
    pass

class Dog(Animal):
    pass

pet = Dog()
print(isinstance(pet, Dog))     # True
print(isinstance(pet, Animal))  # True
print(isinstance(pet, str))     # False
print(type(pet) is Animal)      # False

isinstance checks the class and its base classes. type(obj) is SomeClass tests the exact type. is compares object identity; it does not test an inheritance relationship.

36__str__: readable objects

class Book:
    def __init__(self, title):
        self.title = title

    def __str__(self) -> str:
        return f"Book: {self.title}"

book = Book("The Hobbit")
print(book)        # Book: The Hobbit
text = str(book)
print(type(text))  # <class 'str'>

__str__ supplies the readable text used by str(obj) and print(obj). Return a string; do not just print inside the method. Without a custom implementation, a plain object normally displays a class name and an identifier-like address.

37Composition: HAS-A

class Teacher:
    def __init__(self, name):
        self.name = name

class Course:
    def __init__(self, title, teacher):
        self.title = title
        self.teacher = teacher

teacher = Teacher("Anna")
course = Course("Python", teacher)
print(course.teacher.name)  # Anna

A Course HAS-A Teacher: store an object in an attribute. A Course is not a Teacher, so inheritance is not the relationship here. Follow the references: course → teacher → name. Inheritance instead models IS-A, such as Dog IS-A Animal.

38SQLite · query patterns

-- DB Browser: Open Database → your_database.db → Execute SQL
-- Run one complete query with the ▶ button.

-- Choose columns from a table
SELECT column_a, column_b
FROM table_name;

-- Filter rows: compare a column to a value
SELECT *
FROM table_name
WHERE category = 'value';

-- One value can match either choice
WHERE category IN ('value_a', 'value_b');

-- Group OR conditions before adding AND
WHERE (category = 'value_a' OR category = 'value_b')
  AND amount > 100;

-- Rename a result column; calculate without changing data
SELECT name, unit_price * quantity AS total_value
FROM table_name
WHERE quantity > 0
ORDER BY total_value DESC;

-- Sort then select a range of results
SELECT *
FROM table_name
ORDER BY amount ASC
LIMIT 5 OFFSET 5;

-- _ means exactly one character in LIKE
WHERE first_name LIKE '____';

These are patterns, not answers to a particular database. Replace table_name and column names with names from Database Structure. SELECT reads data; FROM chooses the table; WHERE filters; ORDER BY sorts; LIMIT restricts results; OFFSET skips earlier rows. Run one complete SELECT statement at a time.
